tblContainers_BEFORE_UPDATE BEFORE UPDATE ON tblcontainers
BEGIN

	DECLARE rowCount INT;
    DECLARE sourceIPcount INT;
    
    SELECT count(`tsv`.`originIP`) INTO sourceIPcount FROM `production`.`tblAtERPStateVector` tsv
		WHERE `tsv`.`originIP` = SUBSTRING_INDEX(USER(), '@', -1)
        AND `tsv`.`source` = 'CONTAINERS' AND `tsv`.`action` = 'ACTIVECHECKLOCK';
        
	IF ( sourceIPcount = 0 ) THEN
    
		SET rowCount = (SELECT count(tpd.`id`) FROM `production`.`tblproducts` tpd WHERE (tpd.`srccontainer` = old.`id` OR tpd.`destcontainer` = old.`id`));

		IF
			NOT (new.`container` <=> old.`container`) OR
            NOT (new.`externalcode` <=> old.`externalcode`)
		THEN
		
			SELECT Count(tpd.`id`) INTO rowCount FROM `production`.`tblproducts` tpd
				WHERE tpd.`srccontainer` = old.`id` OR tpd.`destcontainer` = old.`id`;
				
			IF rowCount > 0 THEN
			
				SIGNAL SQLSTATE '45000'
				SET MESSAGE_TEXT = 'Containers - Selected Container is Active and Can Not be Changed. Has Items Under.';
			
			END IF;
			
			SELECT Count(tsm.`id`) INTO rowCount FROM `production`.`tblstockmovement` tsm
				WHERE tsm.`srccontainer` = old.`id` OR tsm.`destcontainer` = old.`id`;
				
			IF rowCount > 0 THEN
			
				SIGNAL SQLSTATE '45000'
				SET MESSAGE_TEXT = 'Containers - Selected Container Can Not be Changed. Has Recorded Transactions';
			
			END IF;
    
		END IF;
    
    END IF;

END

tblContainers_BEFORE_DELETE BEFORE DELETE ON tblcontainers

	SELECT Count(id) INTO rowCount FROM `production`.`tblproducts`
		WHERE `tblproducts`.`srccontainer` = old.id OR `tblproducts`.`destcontainer` = old.id;
		
	IF rowCount > 0 THEN

		SIGNAL SQLSTATE '45000'

	END IF;
END
